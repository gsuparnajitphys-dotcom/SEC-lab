\subsection*{Question 1}
\addcontentsline{toc}{subsection}{Question 1}

Plot the following functions for $-10 \le x \le 10$ on the same graph using GNUPLOT. Set $1000$ samples. Take the range of $y$-values from $-4$ to $+4$. Mark each curve with different line styles and line widths. Label $x$-axis and $y$-axis as \texttt{'x-values'} and \texttt{'y-values'} respectively. Set the title of the plot as \texttt{'Function plots'}. Save the plot in a suitable format.
\[
f_1(x) = \coth(x) - \frac{1}{x}, \quad f_2(x) = e^{-\sin(x)}, \quad f_3(x) = x|x|
\]
\newline
\textbf{Sol:} The GNU-Plot code for the solution of this problem is given below.

\begin{lstlisting}[language=gnuplot]
reset
set term svg background "white"
set ou "/home/suparnajit2005/GitHub/SEC-lab/GNU-plot/Lab-files/Output/1/Q_xxi_ou.svg"

set title "Function plots"
set grid
set xzeroaxis linestyle 1 lc "black"
set yzeroaxis linestyle 1 lc "black"
set xrange [-10:10]
set yrange [-4:4]
set xlabel "x-values"
set ylabel "y-values"
set samples 1000

f1(x)=(1/(tanh(x)))-(1/x)
f2(x)=exp(-sin(x))
f3(x)=x*abs(x)

plot f1(x) lw 1 lc "#5f0f40" dt 1, f2(x) lw 2 lc "#fb8b24" dt 2, f3(x) lw 3 lc "#9a031e" dt 3

\end{lstlisting}

\subsection*{Question 2}
\addcontentsline{toc}{subsection}{Question 2}
Plot the following functions for $0 \le x \le \pi$ on the same graph using GNUPLOT. Set $1000$ samples.Mark each curve with different line style and line widths. Label x-axis and y-axis as 'x-values' and 'y-values' respectively. Set the title of the plot 'Function plots'. Save the plot in a suitable format.
\[
f_1(x) = \sin^2(\sqrt{x}), \quad f_2(x) = e^{-x/2}, \quad f_3(x) = 12.34x^{5/3}
\]
\vspace{10pt}

\noindent \textbf{Sol:} The GNU-Plot code for the solution of this problem is given below.
\vspace{10pt}
\begin{lstlisting}[language=gnuplot]
reset

set ou "/home/suparnajit2005/GitHub/SEC-lab/GNU-plot/Lab-files/Output/1/Q_xx_ou.svg"

set title "Function plots"
set grid
set xzeroaxis linestyle 1 lc "black"
set yzeroaxis linestyle 1 lc "black"
set xrange [0:pi]
set yrange [0:8]
set xtics 0.5 
set ytics 1
set xlabel "x-values"
set ylabel "y-values"

f1(x)=sin(sqrt(x))**2
f2(x)=exp(-x/2)
f3(x)=12.34*(x**(5/3))

set samples 1000
plot f1(x) lw 1 dt 1 lc "#780000", f2(x) lw 2 dt 2 lc "#c1121f", f3(x) lw 3 dt 3 lc "#003049"

\end{lstlisting}

\subsection*{Question 3}
\addcontentsline{toc}{subsection}{Question 3}

Define three functions in the given range:
\[
\begin{aligned}
f_1(x) &= \ln\left(\frac{1}{x}\right), \quad &x > 0.1, \\
f_2(x) &= 3x^2 \sin x, \quad &-1 < x < 0.1, \\
f_3(x) &= \cos 3x, \quad &x < -1.
\end{aligned}
\]
Set the range of $x$ values from $-3$ to $+3$ and the range of $y$ values from $-2.5$ to $+2.5$. Show both the axes on the plot. Set $y$ label ``Piecewise continuous function''. Set ticks of the $y$ axis at an interval of $0.5$. Plot the three functions and store the plot in a file and print it.

\vspace{0.5em}
\noindent \textbf{Sol:} The GNU-Plot code for the solution of this problem is given below.

\begin{lstlisting}[language=gnuplot]
reset
set term svg background "white"
set ou "/home/suparnajit2005/GitHub/SEC-lab/GNU-plot/Lab-files/Output/1/Q_xiv_ou.svg"

set grid

set xzeroaxis linestyle 1 lc "black"
set yzeroaxis linestyle 1 lc "black"

set ytics 0.5
   
set xrange [-3:3]
set yrange [-2.5:2.5]

set ylabel "Piecewise continuous function"


f1(x)=(x>0.1)?(log(1/x)):NaN #Not as Number
f2(x)=(x<0.1 && x>-1)?(3*(x**2)*sin(x)):NaN
f3(x)=(x<-1)?(cos(3*x)):NaN

plot f1(x) lc "#9b2226" lw 2 , f2(x) lc "#bb3e03" lw 2 , f3(x) lc "#ee9b00" lw 2
\end{lstlisting}

Plot

$$2x \sin 2x, \quad \frac{1}{2}x \sinh \left(\frac{x}{2}\right)$$

\noindent on the same graph. The line width of the two curves to be different. Set the range of $x$ values from $-\pi$ to $+\pi$ and the range of $y$ values from $-5$ to $5$. Show both the axes on the plot. Ticks of $x$ axis will be at interval of $0.5$. The label of $x$ and $y$ axis to be "x values" and "y values" respectively. Store the plot in a file and print it.


\noindent \textbf{Sol:} The GNU-Plot code for the solution of this problem is given below.

\vspace{10pt}

% --- GNUPLOT Code Area (Blank) ---
\begin{lstlisting}[language=gnuplot]
